# React → MP4 Converter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir una app de escritorio Windows (Electron + TypeScript) que convierte proyectos React a video MP4, con GUI, Batch Queue y CLI.

**Architecture:** Electron con main process como orquestador de cuatro módulos independientes (`builder`, `capturer`, `encoder`, `queue`). La GUI usa React + Tailwind en el renderer process, comunicada via IPC con contextBridge. El CLI reutiliza los mismos módulos sin levantar ventana.

**Tech Stack:** Electron 29, TypeScript 5, React 18, Tailwind CSS 3, fluent-ffmpeg + ffmpeg-static, express (serve estático), portfinder, commander (CLI), electron-vite (build), electron-builder (packaging), vitest (tests).

```typescript
// src/main/builder.ts
import { spawn } from 'child_process'
import * as http from 'http'
import * as path from 'path'
import * as fs from 'fs'
import express from 'express'
import portfinder from 'portfinder'

export interface BuildResult {
  url: string
  port: number
  stop: () => Promise<void>
}

export async function buildAndServe(
  projectPath: string,
  onProgress: (msg: string) => void
): Promise<BuildResult> {
  if (!fs.existsSync(path.join(projectPath, 'package.json'))) {
    throw new Error(`No package.json found in ${projectPath}`)
  }

  onProgress('Running npm install...')
  await runCommand('npm', ['install'], projectPath)

  onProgress('Running npm run build...')
  await runCommand('npm', ['run', 'build'], projectPath)

  const distPath = path.join(projectPath, 'dist')
  if (!fs.existsSync(distPath)) {
    throw new Error(`Build completed but no dist/ folder found in ${projectPath}`)
  }

  const port = await portfinder.getPortPromise({ port: 40000 })
  const app = express()
  app.use(express.static(distPath))
  // SPA fallback
  app.get('*', (_req, res) => res.sendFile(path.join(distPath, 'index.html')))

  const server = http.createServer(app)
  await new Promise<void>((resolve) => server.listen(port, resolve))

  return {
    url: `http://localhost:${port}`,
    port,
    stop: () => new Promise<void>((resolve) => server.close(() => resolve())),
  }
}

function runCommand(cmd: string, args: string[], cwd: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const proc = spawn(cmd, args, { cwd, shell: true })
    let stderr = ''
    proc.stderr?.on('data', (d: Buffer) => (stderr += d.toString()))
    proc.on('close', (code: number) => {
      if (code !== 0) reject(new Error(`${cmd} ${args.join(' ')} failed:\n${stderr}`))
      else resolve()
    })
  })
}
```

```typescript
// src/main/capturer.ts
import { BrowserWindow } from 'electron'
import * as path from 'path'
import * as fs from 'fs'
import * as os from 'os'

export interface CaptureOptions {
  url: string
  width: number
  height: number
  fps: number
  duration: number
  onProgress: (frame: number, total: number) => void
}

export interface CaptureResult {
  framesDir: string
  frameCount: number
}

export async function captureFrames(options: CaptureOptions): Promise<CaptureResult> {
  const { url, width, height, fps, duration, onProgress } = options
  const totalFrames = Math.round(fps * duration)
  const intervalMs = 1000 / fps

  const framesDir = path.join(os.tmpdir(), `react-to-mp4-${Date.now()}`)
  fs.mkdirSync(framesDir, { recursive: true })

  const win = new BrowserWindow({
    width,
    height,
    show: false,
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
      offscreen: false,
    },
  })

  try {
    await loadPage(win, url)
    // Allow animations to initialize
    await sleep(500)

    for (let i = 0; i < totalFrames; i++) {
      const start = Date.now()

      const image = await win.webContents.capturePage({
        x: 0,
        y: 0,
        width,
        height,
      })

      const framePath = path.join(framesDir, `frame-${String(i).padStart(6, '0')}.png`)
      fs.writeFileSync(framePath, image.toPNG())
      onProgress(i + 1, totalFrames)

      const elapsed = Date.now() - start
      const wait = intervalMs - elapsed
      if (wait > 0) await sleep(wait)
    }

    return { framesDir, frameCount: totalFrames }
  } catch (err) {
    // Clean up frames on error
    fs.rmSync(framesDir, { recursive: true, force: true })
    throw err
  } finally {
    win.destroy()
  }
}

function loadPage(win: BrowserWindow, url: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const timeout = setTimeout(
      () => reject(new Error('Page load timeout (30s)')),
      30_000
    )
    win.webContents.once('did-finish-load', () => {
      clearTimeout(timeout)
      resolve()
    })
    win.webContents.once('did-fail-load', (_e, _code, desc) => {
      clearTimeout(timeout)
      reject(new Error(`Page load failed: ${desc}`))
    })
    win.loadURL(url)
  })
}

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms))
}
```

```typescript
// src/main/encoder.ts
import ffmpeg from 'fluent-ffmpeg'
import * as path from 'path'
import * as fs from 'fs'
import ffmpegPath from 'ffmpeg-static'

ffmpeg.setFfmpegPath(ffmpegPath!)

export interface EncodeOptions {
  framesDir: string
  outputPath: string
  fps: number
  width: number
  height: number
  onProgress: (percent: number) => void
}

export function encodeToMp4(options: EncodeOptions): Promise<void> {
  const { framesDir, outputPath, fps, width, height, onProgress } = options

  return new Promise((resolve, reject) => {
    const cleanup = () => fs.rmSync(framesDir, { recursive: true, force: true })

    ffmpeg()
      .input(path.join(framesDir, 'frame-%06d.png'))
      .inputFPS(fps)
      .videoCodec('libx264')
      .outputOptions([
        '-pix_fmt yuv420p',
        `-vf scale=${width}:${height}`,
        '-preset fast',
        '-crf 18',
      ])
      .fps(fps)
      .output(outputPath)
      .on('progress', (p: { percent?: number }) => onProgress(Math.round(p.percent ?? 0)))
      .on('end', () => { cleanup(); resolve() })
      .on('error', (err: Error) => { cleanup(); reject(err) })
      .run()
  })
}
```

```typescript
// src/main/queue.ts
import { EventEmitter } from 'events'
import { ConversionJob, ConversionOptions } from './types'
import { randomUUID } from 'crypto'

export class ConversionQueue extends EventEmitter {
  private jobs: ConversionJob[] = []
  private running = false

  addJob(inputPath: string, outputPath: string, options: ConversionOptions): ConversionJob {
    const job: ConversionJob = {
      id: randomUUID(),
      inputPath,
      outputPath,
      options,
      status: 'pending',
      progress: 0,
    }
    this.jobs.push(job)
    this.emit('job-updated', job)
    return job
  }

  getJobs(): ConversionJob[] {
    return [...this.jobs]
  }

  clearCompleted(): void {
    this.jobs = this.jobs.filter((j) => j.status !== 'completed')
    this.emit('job-updated', null)
  }

  async start(
    runJob: (job: ConversionJob, onProgress: (p: number) => void) => Promise<void>
  ): Promise<void> {
    if (this.running) return
    this.running = true

    try {
      while (this.running) {
        const next = this.jobs.find((j) => j.status === 'pending')
        if (!next) break

        next.status = 'in_progress'
        this.emit('job-updated', { ...next })

        try {
          await runJob(next, (progress) => {
            next.progress = progress
            this.emit('job-updated', { ...next })
          })
          next.status = 'completed'
          next.progress = 100
        } catch (err) {
          next.status = 'failed'
          next.error = err instanceof Error ? err.message : String(err)
        }

        this.emit('job-updated', { ...next })
      }
    } finally {
      this.running = false
      this.emit('queue-idle')
    }
  }

  stop(): void {
    this.running = false
  }
}
```

```typescript
// src/main/ipc.ts
import { ipcMain, dialog, BrowserWindow, app } from 'electron'
import * as path from 'path'
import { ConversionJob, ConversionOptions } from './types'
import { buildAndServe, BuildResult } from './builder'
import { captureFrames } from './capturer'
import { encodeToMp4 } from './encoder'
import { ConversionQueue } from './queue'

const queue = new ConversionQueue()
let currentBuildResult: BuildResult | null = null
let cancelRequested = false

async function runConversion(
  job: ConversionJob,
  onProgress: (p: number) => void,
  win: BrowserWindow
): Promise<void> {
  cancelRequested = false

  // Phase 1: Build (0-30%)
  const sendProgress = (phase: string, message: string, percent: number) => {
    win.webContents.send('convert:progress', { phase, message, percent })
  }

  sendProgress('building', 'Running npm install & build...', 0)
  const buildResult = await buildAndServe(job.inputPath, (msg) => {
    sendProgress('building', msg, 10)
  })
  currentBuildResult = buildResult
  onProgress(30)

  try {
    if (cancelRequested) throw new Error('Cancelled')

    // Phase 2: Capture frames (30-70%)
    sendProgress('capturing', 'Capturing frames...', 30)
    const { framesDir } = await captureFrames({
      url: buildResult.url,
      width: job.options.width,
      height: job.options.height,
      fps: job.options.fps,
      duration: job.options.duration,
      onProgress: (frame, total) => {
        const pct = 30 + Math.round((frame / total) * 40)
        onProgress(pct)
        sendProgress('capturing', `Capturing frame ${frame}/${total}`, pct)
      },
    })

    if (cancelRequested) throw new Error('Cancelled')

    // Phase 3: Encode (70-100%)
    sendProgress('encoding', 'Encoding MP4...', 70)
    const outputPath = path.join(job.outputPath, job.options.outputName)
    await encodeToMp4({
      framesDir,
      outputPath,
      fps: job.options.fps,
      width: job.options.width,
      height: job.options.height,
      onProgress: (pct) => {
        const total = 70 + Math.round(pct * 0.3)
        onProgress(total)
        sendProgress('encoding', `Encoding... ${pct}%`, total)
      },
    })
    onProgress(100)
  } finally {
    await buildResult.stop()
    currentBuildResult = null
  }
}

export function registerIpcHandlers(win: BrowserWindow): void {
  // Dialogs
  ipcMain.handle('dialog:selectFolder', async () => {
    const result = await dialog.showOpenDialog(win, {
      properties: ['openDirectory'],
    })
    return result.canceled ? null : result.filePaths[0]
  })

  // Single conversion
  ipcMain.handle(
    'convert:start',
    async (_event, inputPath: string, outputPath: string, options: ConversionOptions) => {
      const job: ConversionJob = {
        id: 'single',
        inputPath,
        outputPath,
        options,
        status: 'in_progress',
        progress: 0,
      }
      try {
        await runConversion(job, (p) => { job.progress = p }, win)
        win.webContents.send('convert:done')
      } catch (err) {
        win.webContents.send('convert:done', err instanceof Error ? err.message : String(err))
      }
    }
  )

  ipcMain.on('convert:cancel', () => {
    cancelRequested = true
    currentBuildResult?.stop()
  })

  // Batch
  ipcMain.handle(
    'batch:add',
    (_event, inputPath: string, outputPath: string, options: ConversionOptions) => {
      const job = queue.addJob(inputPath, outputPath, options)
      win.webContents.send('batch:updated', queue.getJobs())
      return job
    }
  )

  ipcMain.on('batch:start', () => {
    queue.start((job, onProgress) => runConversion(job, onProgress, win))
    queue.on('job-updated', () => win.webContents.send('batch:updated', queue.getJobs()))
  })

  ipcMain.on('batch:stop', () => queue.stop())

  ipcMain.on('batch:clear', () => {
    queue.clearCompleted()
    win.webContents.send('batch:updated', queue.getJobs())
  })
}
```

```typescript
// src/renderer/src/hooks/useConversion.ts
import { useState, useEffect, useCallback } from 'react'
import { ConversionOptions, ProgressData, DEFAULT_OPTIONS } from '../../../main/types'
import { ipc } from '../ipc'

type Phase = 'idle' | 'converting' | 'done' | 'error'

export function useConversion() {
  const [inputPath, setInputPath] = useState('')
  const [outputPath, setOutputPath] = useState('')
  const [options, setOptions] = useState<ConversionOptions>({ ...DEFAULT_OPTIONS })
  const [phase, setPhase] = useState<Phase>('idle')
  const [progress, setProgress] = useState<ProgressData>({ phase: 'building', message: '', percent: 0 })
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    const unsubProgress = ipc.onConvertProgress((data) => {
      setProgress(data)
    })
    const unsubDone = ipc.onConvertDone((err) => {
      if (err) { setPhase('error'); setError(err) }
      else setPhase('done')
    })
    return () => { unsubProgress(); unsubDone() }
  }, [])

  const selectInput = useCallback(async () => {
    const p = await ipc.selectFolder()
    if (p) setInputPath(p)
  }, [])

  const selectOutput = useCallback(async () => {
    const p = await ipc.selectFolder()
    if (p) setOutputPath(p)
  }, [])

  const start = useCallback(async () => {
    setPhase('converting')
    setError(null)
    setProgress({ phase: 'building', message: 'Iniciando...', percent: 0 })
    try {
      await ipc.convertStart(inputPath, outputPath, options)
    } catch (e) {
      setPhase('error')
      setError(e instanceof Error ? e.message : String(e))
    }
  }, [inputPath, outputPath, options])

  const cancel = useCallback(() => {
    ipc.convertCancel()
    setPhase('idle')
  }, [])

  const addToBatch = useCallback(async () => {
    if (!inputPath || !outputPath) return
    await ipc.batchAdd(inputPath, outputPath, options)
  }, [inputPath, outputPath, options])

  return {
    inputPath, outputPath, options, phase, progress, error,
    setOptions, selectInput, selectOutput, start, cancel, addToBatch,
  }
}
```

```typescript
// src/renderer/src/hooks/useBatchQueue.ts
import { useState, useEffect } from 'react'
import { ConversionJob } from '../../../main/types'
import { ipc } from '../ipc'

export function useBatchQueue() {
  const [jobs, setJobs] = useState<ConversionJob[]>([])
  const [running, setRunning] = useState(false)

  useEffect(() => {
    const unsub = ipc.onBatchUpdated((updatedJobs) => {
      setJobs(updatedJobs)
      const hasActive = updatedJobs.some((j) => j.status === 'in_progress')
      setRunning(hasActive)
    })
    return unsub
  }, [])

  const startAll = () => {
    setRunning(true)
    ipc.batchStart()
  }
  const stop = () => {
    setRunning(false)
    ipc.batchStop()
  }
  const clear = () => ipc.batchClear()

  return { jobs, running, startAll, stop, clear }
}
```

```typescript
async function convertSingle(
  inputPath: string,
  outputPath: string,
  opts: ConversionOptions
): Promise<void> {
  console.log(`\n🚀 Convirtiendo: ${path.basename(inputPath)}`)

  // Build
  process.stdout.write('📦 Building React project...')
  const build = await buildAndServe(path.resolve(inputPath), (msg) => {
    process.stdout.write(`\r📦 ${msg}`)
  })
  console.log('\r✅ Build completado                ')

  try {
    // Dynamic import de puppeteer (puede no estar instalado en todos los entornos)
    let puppeteer: any
    try {
      puppeteer = await import('puppeteer')
    } catch {
      throw new Error(
        'puppeteer no encontrado. Instálalo con: npm install puppeteer\n' +
        '(Solo requerido para el CLI. La GUI usa el Chromium de Electron.)'
      )
    }

    const browser = await puppeteer.default.launch({ headless: true })
    const page = await browser.newPage()
    await page.setViewport({ width: opts.width, height: opts.height })
    await page.goto(build.url, { waitUntil: 'networkidle2', timeout: 30000 })
    await new Promise((r) => setTimeout(r, 500))

    const totalFrames = Math.round(opts.fps * opts.duration)
    const intervalMs = 1000 / opts.fps
    const os = await import('os')
    const framesDir = path.join(os.tmpdir(), `react-to-mp4-cli-${Date.now()}`)
    fs.mkdirSync(framesDir, { recursive: true })

    console.log(`📸 Capturando ${totalFrames} frames a ${opts.fps}fps...`)
    for (let i = 0; i < totalFrames; i++) {
      const start = Date.now()
      await page.screenshot({
        path: path.join(framesDir, `frame-${String(i).padStart(6, '0')}.png`),
        clip: { x: 0, y: 0, width: opts.width, height: opts.height },
      })
      const pct = Math.round(((i + 1) / totalFrames) * 100)
      process.stdout.write(`\r  [${pct.toString().padStart(3)}%] Frame ${i + 1}/${totalFrames}`)
      const wait = intervalMs - (Date.now() - start)
      if (wait > 0) await new Promise((r) => setTimeout(r, wait))
    }
    console.log('\r✅ Frames capturados                    ')
    await browser.close()

    // Encode
    const outputFile = path.join(path.resolve(outputPath), opts.outputName)
    process.stdout.write('🎬 Encoding MP4...')
    await encodeToMp4({
      framesDir,
      outputPath: outputFile,
      fps: opts.fps,
      width: opts.width,
      height: opts.height,
      onProgress: (pct) => process.stdout.write(`\r🎬 Encoding... ${pct}%`),
    })
    console.log(`\r✅ MP4 guardado en: ${outputFile}`)
  } finally {
    await build.stop()
  }
}

  .name('react-to-mp4')
  .description('Convert React animations to MP4')
  .version('1.0.0')

program
  .option('-i, --input <path>', 'React project folder')
  .option('-o, --output <path>', 'Output folder', '.')
  .option('-w, --width <px>', 'Video width', String(DEFAULT_OPTIONS.width))
  .option('-h, --height <px>', 'Video height', String(DEFAULT_OPTIONS.height))
  .option('-f, --fps <n>', 'Frames per second', String(DEFAULT_OPTIONS.fps))
  .option('-d, --duration <s>', 'Duration in seconds', String(DEFAULT_OPTIONS.duration))
  .option('-n, --name <file>', 'Output filename', DEFAULT_OPTIONS.outputName)
  .option('-b, --batch <json>', 'Batch mode: path to jobs JSON file')
  .action(async (options) => {
    try {
      if (options.batch) {
        const jobs = JSON.parse(fs.readFileSync(path.resolve(options.batch), 'utf8'))
        for (const job of jobs) {
          const opts: ConversionOptions = {
            width: job.width ?? DEFAULT_OPTIONS.width,
            height: job.height ?? DEFAULT_OPTIONS.height,
            fps: job.fps ?? DEFAULT_OPTIONS.fps,
            duration: job.duration ?? DEFAULT_OPTIONS.duration,
            outputName: job.name ?? DEFAULT_OPTIONS.outputName,
          }
          await convertSingle(job.input, job.output ?? '.', opts)
        }
      } else {
        if (!options.input) {
          console.error('❌ Error: --input es requerido')
          process.exit(1)
        }
        const opts: ConversionOptions = {
          width: parseInt(options.width),
          height: parseInt(options.height),
          fps: parseInt(options.fps),
          duration: parseFloat(options.duration),
          outputName: options.name,
        }
        await convertSingle(options.input, options.output, opts)
      }
      process.exit(0)
    } catch (err) {
      console.error('\n❌ Error:', err instanceof Error ? err.message : err)
```
